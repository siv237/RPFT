\chapter{Минимальная архитектура фермионного резервуара для логарифма определителя KMS}
\label{ch:v9-kms-minimal-fermion-bath}

\section{Минимальный новый носитель}

Physical-loop запретительный результат локализовал нехватку одного determinant degree five.
Минимальная bath поэтому повторяет five-channel type-space:
\begin{equation}
 V_{\rm sys}\simeq\mathbb C^5,
 \qquad V_{\rm bath}\simeq\mathbb C^5,
 \qquad \mathcal F_{\rm sb}=V_{\rm sys}\oplus V_{\rm bath},
 \qquad \dim_{\mathbb C}\mathcal F_{\rm sb}=10.
 \label{eq:v9-kms-fermion-bath-carrier}
\end{equation}
Обе части несут одинаковое type-разложение
\begin{equation}
 V_{\rm sys},V_{\rm bath}=1_s\oplus1_a\oplus3_t.
 \label{eq:v9-kms-fermion-bath-types}
\end{equation}
Тем самым новый носитель минимален по недостающей determinant degree:
\begin{equation}
 \deg\det R_\kappa=1+1+3=5.
 \label{eq:v9-kms-fermion-bath-minimal-degree}
\end{equation}

\section{Typed system--bath coupling}

Gauge, grading и family covariance оставляют diagonal coupling
\begin{equation}
 G=\operatorname{diag}(g_s,g_a,g_t,g_t,g_t),
 \qquad g_s,g_a,g_t>0,
 \qquad \rank G=5.
 \label{eq:v9-kms-fermion-bath-coupling}
\end{equation}
Triplet block пропорционален единице, поэтому
\begin{equation}
 [G,0\oplus0\oplus X_t]=0,
 \qquad X_t\in\mathfrak{so}(3).
 \label{eq:v9-kms-fermion-bath-family-covariance}
\end{equation}

\section{Coupled Hermitian kernel}

Минимальный Euclidean Hermitian kernel равен
\begin{equation}
 K_H=\begin{pmatrix}R_\theta&G\\G&R_\kappa\end{pmatrix}.
 \label{eq:v9-kms-fermion-bath-hermitian-kernel}
\end{equation}
Его determinant факторизуется поканально:
\begin{equation}
 \det K_H=
 (r_{\theta s}r_{\kappa s}-g_s^2)
 (r_{\theta a}r_{\kappa a}-g_a^2)
 (r_{\theta t}r_{\kappa t}-g_t^2)^3.
 \label{eq:v9-kms-fermion-bath-full-determinant}
\end{equation}
После интегрирования bath system-kernel становится Schur complement
\begin{equation}
 K_{\rm eff}=R_\theta-GR_\kappa^{-1}G,
 \qquad
 \det K_H=\det R_\kappa\det K_{\rm eff}.
 \label{eq:v9-kms-fermion-bath-schur}
\end{equation}
В type-базисе
\begin{equation}
 K_{\rm eff}=\operatorname{diag}\!\left(
 r_{\theta s}-\frac{g_s^2}{r_{\kappa s}},
 r_{\theta a}-\frac{g_a^2}{r_{\kappa a}},
 \left(r_{\theta t}-\frac{g_t^2}{r_{\kappa t}}\right)I_3
 \right).
 \label{eq:v9-kms-fermion-bath-schur-diagonal}
\end{equation}
Отношение к target determinant равно
\begin{equation}
 \frac{\det K_H}{\det R_\theta\det R_\kappa}
 =\left(1-\frac{g_s^2}{r_{\theta s}r_{\kappa s}}\right)
  \left(1-\frac{g_a^2}{r_{\theta a}r_{\kappa a}}\right)
  \left(1-\frac{g_t^2}{r_{\theta t}r_{\kappa t}}\right)^3.
 \label{eq:v9-kms-fermion-bath-ratio}
\end{equation}
Следовательно, nonzero Hermitian coupling создаёт self-energy, но меняет
сам логарифм определителя functional.

\section{Exact positive witness}

Выберем
\begin{equation}
 R_\theta=R_\kappa=2I_5,
 \qquad G=I_5.
 \label{eq:v9-kms-fermion-bath-witness}
\end{equation}
Тогда coupled kernel строго положителен:
\begin{equation}
 \Spec K_H^{(w)}=\{1^{(5)},3^{(5)}\}.
 \label{eq:v9-kms-fermion-bath-witness-spectrum}
\end{equation}
Но его determinant равен
\begin{equation}
 \det K_H^{(w)}=3^5=243,
 \label{eq:v9-kms-fermion-bath-witness-determinant}
\end{equation}
тогда как target на тех же двух blocks есть
\begin{equation}
 \det R_\theta\det R_\kappa=2^{10}=1024.
 \label{eq:v9-kms-fermion-bath-witness-target}
\end{equation}
Точный defect и ratio:
\begin{equation}
 1024-243=781>0,
 \qquad
 \frac{\det K_H^{(w)}}{\det R_\theta\det R_\kappa}
 =\frac{243}{1024}.
 \label{eq:v9-kms-fermion-bath-witness-defect}
\end{equation}

\section{Zero-coupling face}

При $G=0$ восстанавливается block direct sum
\begin{equation}
 K_0=R_\theta\oplus R_\kappa,
 \qquad
 \det K_0=\det R_\theta\det R_\kappa.
 \label{eq:v9-kms-fermion-bath-zero-face}
\end{equation}
Однако induced system self-energy одновременно исчезает:
\begin{equation}
 \Sigma_{\rm sys}=GR_\kappa^{-1}G=0.
 \label{eq:v9-kms-fermion-bath-zero-self-energy}
\end{equation}
Decoupled bath даёт формальный determinant factor, но не может быть
microscopic origin conductance.

\section{Causal route следующего гейта}

Единственный очевидный finite-block способ сохранить determinant при
nonzero off-diagonal entry есть triangular kernel
\begin{equation}
 K_{\triangle}=\begin{pmatrix}R_\theta&G\\0&R_\kappa\end{pmatrix},
 \qquad \det K_{\triangle}=\det R_\theta\det R_\kappa.
 \label{eq:v9-kms-fermion-bath-triangular-candidate}
\end{equation}
Но он не является Euclidean Hermitian operator. Его можно рассматривать
только после явного Schwinger--Keldysh causal doubling, noise block,
normalization и KMS symmetry; это содержание следующего гейта.
