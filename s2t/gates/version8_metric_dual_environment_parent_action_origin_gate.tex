\chapter{Происхождение метрически двойственной среды из родительского действия}
\label{ch:v8-metric-dual-environment-parent-action-origin-gate}

\section{Проверяемое усиление}

Предыдущая глава получила метрику межсекторных скоростей
\begin{equation}
 R_{\rm tr}=K_B^{-1}=\frac13I_{12}
 \label{eq:v8-riesz-cross-rate}
\end{equation}
при условии, что среда скачков является метрически двойственным модулем к
полевому межсекторному модулю. Теперь проверим, следует ли это условие из уже
имеющейся квадратичной части действия.

На вещественных координатах $x\in E_{\rm cross}$ полевой суперслед задаёт
\begin{equation}
 S_{\rm field}^{(2)}(x)=\frac12x^TK_Bx,
 \qquad K_B=3I_{12}.
 \label{eq:v8-cross-field-parent-restriction}
\end{equation}
Чтобы действие определяло среда, к нему требуется присоединить переменные
$\xi$ и положительную ковариацию среды $R$. Квадратичный квадратичное дополнение имеет вид
\begin{equation}
 S_R^{(2)}(x,\xi)
 =\frac12x^TK_Bx+\frac12\xi^TR^{-1}\xi.
 \label{eq:v8-cross-field-bath-parent-completion}
\end{equation}

\section{Два точных квадратичное дополнение одного действия}

Рассмотрим две положительные калибровочный коммутантные метрики
\begin{equation}
 R_1=\frac13I_{12},
 \qquad
 R_2=\operatorname{diag}\left(
 \frac13I_6,\frac23I_6\right).
 \label{eq:v8-two-bath-rate-completions}
\end{equation}
Они не пропорциональны. Тем не менее оба родительских гессиана
\begin{align}
 \mathcal H_1&=\operatorname{diag}(3I_{12},3I_{12}),\\
 \mathcal H_2&=\operatorname{diag}
 \left(3I_{12},3I_6,\frac32I_6\right)
 \label{eq:v8-two-parent-hessians}
\end{align}
положительны, калибровочно ковариантны и имеют одну и ту же полевую рестрикцию
$K_B$. Размерность минимальной среды скачков также не меняется.

Различие физически видимо уже на генераторе: $R_1$ даёт одинаковые веса
$1/3$ двум межсекторным семействам, а $R_2$ даёт веса $1/3$ и $2/3$. Точный
матричных единиц свидетель различия редуцированных супероператоров равен
$E_{0,8}$. Следовательно, одна полевая часть действия допускает по меньшей
мере две неэквивалентные динамики.

\section{Что именно добавляет Рис-принцип}

Условие метрической двойственности выражается отдельным уравнением
\begin{equation}
 K_BR=I_{12}.
 \label{eq:v8-riesz-reciprocity-condition}
\end{equation}
Поскольку $K_B$ невырождена, оно имеет единственное решение
$R=K_B^{-1}$. Для $R_1$ остаток равен нулю, а для $R_2$ --- нет. Таким
образом, Рис-принцип является достаточным и минимальным селектором формы
скоростной метрики, но он не следует из знания одной рестрикции
\eqref{eq:v8-cross-field-parent-restriction}.

\begin{theorem}[Точная локализация недостающей аксиомы]
Полевой суперслед $K_B=3I_{12}$ вместе с калибровочная ковариантностью и
минимальной размерностью среды не определяет единственную ковариацию среды:
$R_1$ и $R_2$ задают два положительных, не пропорциональных квадратичное дополнение с
одной полевой рестрикцией и разными редуцированными генераторами. Добавочное
Рис-условие $K_BR=I_{12}$ единственно выбирает $R_1=I_{12}/3$.
\end{theorem}

\begin{nogocriterion}[Родительское действие без сектора среды не выводит среду]
Нельзя объявлять метрически двойственную среду следствием существующего
полевого действия, пока в одном функционале не предъявлены переменные среды
или эквивалентное кокасательное/BV-удвоение и не выведено условие $K_BR=I$.
Простое добавление ограничения с множителем Лагранжа было бы корректной
реализацией принципа, но не его независимым выводом.
\end{nogocriterion}

\begin{protocol}
\begin{enumerate}
 \item полевая метрика: \textbf{$K_B=3I_{12}$};
 \item положительных родительское дополнение: \textbf{не менее $2$};
 \item одинаковая полевая рестрикция: \textbf{точно};
 \item калибровочные коммутаторы обеих скоростных метрик: \textbf{нулевые};
 \item квадратичное дополнение пропорциональны: \textbf{нет};
 \item свидетель различных динамик: \textbf{$E_{0,8}$};
 \item Рис-условие: \textbf{$K_BR=I_{12}$};
 \item единственное Рис-решение: \textbf{$I_{12}/3$};
 \item происхождение Рис-условия из старого действия: \textbf{нет};
 \item физический масштаб времени: \textbf{не выведен};
 \item следующий гейт: \textbf{единый кокасательный родитель полного шумового
       модуля либо источник свежих вспомогательных систем}.
\end{enumerate}
\end{protocol}
