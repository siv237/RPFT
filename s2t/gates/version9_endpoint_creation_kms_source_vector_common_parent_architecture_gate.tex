\chapter{Общий родительский функционал источников параметров KMS}
\label{ch:v9-endpoint-kms-source-parent}

\section{Source-carrier}

Для трёх channel types введём вещественное пространство
\begin{equation}
 T_{\rm KMS}=\operatorname{span}_{\mathbb R}\{e_s,e_a,e_t\},
 \qquad
 \mathcal J_{\rm KMS}=T_{\rm KMS}^*\oplus T_{\rm KMS}^*.
 \label{eq:v9-kms-source-parent-carrier}
\end{equation}
Первая копия несёт gap-source $j_\theta$, вторая --- conductance-source
$j_\kappa$. Multiplicity trace задаёт положительную метрику
\begin{equation}
 W=\operatorname{diag}(1,1,3),
 \qquad
 \mathbb W=W\oplus W,
 \qquad
 \det\mathbb W=9.
 \label{eq:v9-kms-source-parent-metric}
\end{equation}

\section{Единый строго выпуклый parent}

На $\mathbb R^3_\theta\oplus\mathbb R^3_\kappa$ определим
\begin{equation}
 S_{\rm KMS}=
 \frac12\boldsymbol\theta^TW\boldsymbol\theta-j_\theta^T\boldsymbol\theta
 +\frac12\boldsymbol\kappa^TW\boldsymbol\kappa-j_\kappa^T\boldsymbol\kappa.
 \label{eq:v9-kms-source-parent-functional}
\end{equation}
Completion of squares даёт
\begin{equation}
 \begin{aligned}
 S_{\rm KMS}
 &=\frac12(\boldsymbol\theta-W^{-1}j_\theta)^T
 W(\boldsymbol\theta-W^{-1}j_\theta)\\
 &\quad+\frac12(\boldsymbol\kappa-W^{-1}j_\kappa)^T
 W(\boldsymbol\kappa-W^{-1}j_\kappa)
 -\frac12j_\theta^TW^{-1}j_\theta
 -\frac12j_\kappa^TW^{-1}j_\kappa .
 \end{aligned}
 \label{eq:v9-kms-source-parent-completion}
\end{equation}
Поэтому минимум единственен:
\begin{equation}
 \boldsymbol\theta_*=W^{-1}j_\theta,
 \qquad
 \boldsymbol\kappa_*=W^{-1}j_\kappa.
 \label{eq:v9-kms-source-parent-minimum}
\end{equation}
Гессиан не зависит от точки:
\begin{equation}
 \Hess S_{\rm KMS}=\mathbb W>0,
 \qquad
 \rank\Hess S_{\rm KMS}=6,
 \qquad
 \det\Hess S_{\rm KMS}=9.
 \label{eq:v9-kms-source-parent-hessian}
\end{equation}

\section{Положительный конус и KMS-map}

Допустимый source-cone равен
\begin{equation}
 \mathcal J_{\rm KMS}^{+}
 =\{(j_\theta,j_\kappa):W^{-1}j_\theta>0,\
 W^{-1}j_\kappa>0\}.
 \label{eq:v9-kms-source-parent-positive-cone}
\end{equation}
Для каждого элемента этого конуса минимум задаёт primitive KMS data
\begin{equation}
 q_\alpha=\exp(-\theta_{\alpha,*}),
 \qquad
 \gamma_\alpha^+=\kappa_{\alpha,*}q_\alpha,
 \qquad
 \gamma_\alpha^-=\kappa_{\alpha,*}>0.
 \label{eq:v9-kms-source-parent-kms-map}
\end{equation}

\section{Вложение в four-slot functional}

Расширенное пространство конфигураций имеет вид
\begin{equation}
 \mathcal X_{\rm IX}^{+}
 =\mathcal X_4\times\mathbb R^3_{\theta,>0}
 \times\mathbb R^3_{\kappa,>0},
 \label{eq:v9-kms-source-parent-common-domain}
\end{equation}
где $\mathcal X_4$ несёт endpoint, $E_*$, $\chi$ и примитив переноса.
Один общий функционал задаётся суммой
\begin{equation}
 S_{\rm IX}^{+}=S_4+S_{\rm KMS}.
 \label{eq:v9-kms-source-parent-common-functional}
\end{equation}
При witness Hessian $\Hess_{E_*,\chi}S_4=\operatorname{diag}(8,8)$
непрерывный гессиан общего родительского функционала равен
\begin{equation}
 \Hess_{\rm cont}S_{\rm IX}^{+}
 =\operatorname{diag}(8,8,1,1,3,1,1,3),
 \quad
 \rank\Hess_{\rm cont}S_{\rm IX}^{+}=8,
 \quad
 \det\Hess_{\rm cont}S_{\rm IX}^{+}=576.
 \label{eq:v9-kms-source-parent-common-hessian}
\end{equation}
Ограничение при $\boldsymbol\theta=\boldsymbol\kappa=0$ буквально
восстанавливает старый функционал $S_4$.

\section{Точный witness}

Выберем ковектор источникаs
\begin{equation}
 j_\theta=(1,2,9)^T,
 \qquad
 j_\kappa=(2,1,6)^T.
 \label{eq:v9-kms-source-parent-witness-sources}
\end{equation}
Тогда
\begin{equation}
 \boldsymbol\theta_*=(1,2,3)^T,
 \qquad
 \boldsymbol\kappa_*=(2,1,2)^T,
 \qquad
 q_*=(e^{-1},e^{-2},e^{-3}),
 \label{eq:v9-kms-source-parent-witness-minimum}
\end{equation}
а минимальное значение KMS-части равно
\begin{equation}
 \min S_{\rm KMS}
 =-\frac12\left(j_\theta^TW^{-1}j_\theta
 +j_\kappa^TW^{-1}j_\kappa\right)
 =-\frac{49}{2}.
 \label{eq:v9-kms-source-parent-witness-value}
\end{equation}

\section{Архитектурный итог}

\begin{equation}
 \begin{aligned}
 N_{\rm source\ parent\ architecture}&=9/9,\\
 N_{\rm conditional\ KMS\ selection}&=6/6,\\
 N_{\rm source\ package\ origin}&=0/2,\\
 N_{\rm source\ component\ origin}&=0/6,\\
 N_{\rm physical\ four\ slot\ parent}&=0/1.
 \end{aligned}
 \label{eq:v9-kms-source-parent-ledger}
\end{equation}

\begin{theorem}[Общий условный source-parent]
Для любых $(j_\theta,j_\kappa)\in\mathcal J_{\rm KMS}^{+}$ функционал
$S_{\rm KMS}$ ограничен снизу, строго выпукл и имеет единственный
положительный минимум. Его сумма с $S_4$ является одним bounded parent на
общем пространство конфигураций и условно выбирает все шесть KMS-параметров.
\end{theorem}

\begin{nogocriterion}[Source-covectors не считаются выведенными]
Архитектура не объясняет значения $j_\theta$ и $j_\kappa$. Пока отсутствует
типизированная карта из четырёх физических slots в эти covectors,
conditional selection $6/6$ не повышает physical parent выше $0/1$.
\end{nogocriterion}

\item conditional KMS selection: \textbf{$6/6$};
\item source-package origin: \textbf{$0/2$};
\item source-component origin: \textbf{$0/6$};
\item следующий гейт: \textbf{four-slot ковектор источника origin}.
\end{enumerate}\end{protocol}
