\section{Базовый gauge-носитель}

Возьмём уже построенный ungraded type/package carrier
\begin{equation}
 E=V_{\rm type}\otimes P_{\rm KMS},
 \qquad \dim_{\mathbb C}E=5\cdot2=10.
 \label{eq:v9-kms-brst-base-carrier}
\end{equation}
Минимальный off-shell contractible quartet над $E$ состоит из
\begin{equation}
 x,b\in E^{\bar0},
 \qquad
 c,\bar c\in E^{\bar1}.
 \label{eq:v9-kms-brst-fields}
\end{equation}
Ghost numbers назначаются как
\begin{equation}
 \operatorname{gh}(x)=\operatorname{gh}(b)=0,
 \qquad
 \operatorname{gh}(c)=1,
 \qquad
 \operatorname{gh}(\bar c)=-1.
 \label{eq:v9-kms-brst-ghost-numbers}
\end{equation}
Таким образом,
\begin{equation}
 \dim X_{\rm BRST}^{\bar0}=20,\qquad
 \dim X_{\rm BRST}^{\bar1}=20,\qquad
 \operatorname{sdim}X_{\rm BRST}=0.
 \label{eq:v9-kms-brst-superdimension}
\end{equation}

\section{Nilpotent differential}

Определим
\begin{equation}
 sx=c,\qquad sc=0,\qquad s\bar c=b,\qquad sb=0.
 \label{eq:v9-kms-brst-differential}
\end{equation}
В порядке $(x,b,c,\bar c)$ его matrix имеет block-форму
\begin{equation}
 Q=
 \begin{pmatrix}
 0&0&0&0\\
 0&0&0&I_{10}\\
 I_{10}&0&0&0\\
 0&0&0&0
 \end{pmatrix}.
 \label{eq:v9-kms-brst-matrix}
\end{equation}
Точные invariants равны
\begin{equation}
 Q^2=0,\qquad
 \rank Q=20,\qquad
 \dim\ker Q=20.
 \label{eq:v9-kms-brst-rank-nullity}
\end{equation}
Поскольку $\operatorname{im}Q\subseteq\ker Q$ и их размерности совпадают,
\begin{equation}
 H^\bullet(X_{\rm BRST},Q)=0.
 \label{eq:v9-kms-brst-zero-cohomology}
\end{equation}
Для ghost-number operator $N_{\rm gh}$ и parity $\Gamma_{\rm BRST}$
выполняются
\begin{equation}
 [N_{\rm gh},Q]=Q,
 \qquad
 \{\Gamma_{\rm BRST},Q\}=0.
 \label{eq:v9-kms-brst-degree-parity}
\end{equation}
Тем самым нечётность $c,\bar c$ теперь является частью BRST architecture,
а не произвольной grading самого determinant carrier.

\section{Gauge-fixing fermion}

Faddeev--Popov operator выбирается равным уже построенному KMS block:
\begin{equation}
 M_{\rm FP}=D_{\rm aux}=R_\theta\oplus R_\kappa,
 \qquad \rank M_{\rm FP}=10.
 \label{eq:v9-kms-brst-fp-operator}
\end{equation}
Для $\alpha>0$ минимальная gauge-fixing fermion имеет вид
\begin{equation}
 \Psi=\bar c^{\,T}\left(D_{\rm aux}x-\frac{\alpha}{2}b\right),
 \qquad \operatorname{gh}\Psi=-1.
 \label{eq:v9-kms-brst-gauge-fermion}
\end{equation}
Её BRST-exact action равен
\begin{equation}
 s\Psi=
 b^T D_{\rm aux}x-\frac{\alpha}{2}b^Tb
 -\bar c^{\,T}D_{\rm aux}c.
 \label{eq:v9-kms-brst-exact-action}
\end{equation}
Ghost block даёт требуемый determinant
\begin{equation}
 \det M_{\rm FP}
 =r_{\theta s}r_{\theta a}r_{\theta t}^3
  r_{\kappa s}r_{\kappa a}r_{\kappa t}^3
 =\det R_\theta\,\det R_\kappa.
 \label{eq:v9-kms-brst-fp-determinant}
\end{equation}
Следовательно, conditional ghost integral воспроизводит
\begin{equation}
 -\log Z_{\rm gh}
 =-\log\det R_\theta-\log\det R_\kappa.
 \label{eq:v9-kms-brst-effective-logdet}
\end{equation}

\section{Минимальность и физическая когомология}

Degree десять требует десять ghost/antighost pairs. Off-shell nilpotence
$s\bar c=b$, $sb=0$ требует по одному even multiplier на gauge condition,
а shift-doublet $sx=c$ --- десять gauge coordinates. Поэтому minimal
off-shell realization имеет
\begin{equation}
 N_x=N_b=N_c=N_{\bar c}=10,
 \qquad
 \dim_{\mathbb C}X_{\rm BRST}=40.
 \label{eq:v9-kms-brst-minimal-count}
\end{equation}
Она присоединяется direct sum к creation-cell:
\begin{equation}
 X_{\rm total}=\mathcal H_{\rm cr}\oplus X_{\rm BRST},
 \qquad
 Q\iota_{\rm phys}=0.
 \label{eq:v9-kms-brst-physical-inclusion}
\end{equation}
Так как auxiliary cohomology нулевая,
\begin{equation}
 H^\bullet(X_{\rm total},Q)
 \simeq\mathcal H_{\rm cr},
 \qquad N_{\rm new\ physical\ states}=0.
 \label{eq:v9-kms-brst-physical-cohomology}
\end{equation}

\end{equation}
