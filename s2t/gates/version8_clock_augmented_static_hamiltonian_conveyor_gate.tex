\chapter{Часовое расширение стационарного гамильтониана конвейера}
\label{ch:v8-clock-augmented-static-hamiltonian-conveyor-gate}

Предыдущий гейт исключил стационарный логарифм на минимальном двухполосном
носителе, но оставил открытым расширение регистром часов. Проверим отдельно
конечный одноразовый подъём и равномерный термодинамический конвейер.

\section{Точные трёхсостоянийные часы истории}

Для двух слоёв $W_0,W_1$ введём состояния часов $|0\rangle,|1\rangle,|2\rangle$
и гамильтониан
\begin{equation}
 H_{\rm hist}=\sqrt2\bigl(
 |1\rangle\langle0|\otimes W_0+|2\rangle\langle1|\otimes W_1+\mathrm{h.c.}
 \bigr).
\end{equation}
После блочно-диагонального преобразования истории он сводится к матрице
\begin{equation}
 A=\begin{pmatrix}0&\sqrt2&0\\ \sqrt2&0&\sqrt2\\0&\sqrt2&0\end{pmatrix},
 \qquad A^3=4A.
\end{equation}
Поэтому при $\tau=\pi/2$
\begin{equation}
 e^{-i\tau H_{\rm hist}}
 (|0\rangle\otimes|\psi\rangle)
 =-|2\rangle\otimes W_1W_0|\psi\rangle.
\end{equation}

\begin{theorem}[Точный конечный подъём часами]
Двухслойный конвейер имеет точную одноразовую реализацию стационарным
гамильтонианом истории на трёхсостоянийном часовом носителе.
\end{theorem}

\section{Почему это ещё не локальный бесконечный конвейер}

Запись выше использует $W_0,W_1$ как целые слои и потому не является
локальной на исходной бесконечной цепи. На кольце длины $L$ локальная
сериализация содержит $T=2L$ переходов. Идеальный перенос по цепи истории
требует связей
\begin{equation}
 J_t=\sqrt{(t+1)(T-t)},\qquad J_{\max}\geq T/2.
\end{equation}
При нормировке $\|J\|\leq1$ время переноса удовлетворяет
\begin{equation}
 \tau_{\rm tr}\geq\frac{\pi T}{4}=\frac{\pi L}{2}.
\end{equation}
Следовательно, точный конечный подъём либо использует растущую силу связей,
либо требует времени, растущего с объёмом.

\begin{nogocriterion}[Граница часового подъёма]
Гамильтониан истории не выводит равномерный по объёму локальный конвейер с
ограниченной силой взаимодействия и фиксированной длительностью такта.
Кроме того, часы заканчивают шаг в $|2\rangle$, а не возвращаются в исходное
состояние; бесконечная повторяемость требует нового носителя или сброса.
Этот вывод относится к явному классу гамильтонианов истории и не исключает
все возможные взаимодействующие часовые модели.
\end{nogocriterion}

\begin{protocol}
\begin{enumerate}
 \item размерность минимальных часов истории: \textbf{$3$};
 \item точное время переноса: \textbf{$\pi/2$};
 \item конечное выполнение $W_1W_0$: \textbf{точно};
 \item возврат часов в $|0\rangle$: \textbf{нет};
 \item локальная сериализация на кольце: \textbf{$T=2L$};
 \item максимальная ненормированная связь: \textbf{не меньше $T/2$};
 \item время при ограниченных связях: \textbf{не меньше $\pi L/2$};
 \item равномерный термодинамический такт: \textbf{не выведен};
 \item следующий гейт: \textbf{термодинамический предел автономных часов с ограниченной силой}.
\end{enumerate}
\end{protocol}
