\chapter{Двухрезервуарный обратный отклик третьей поправки}
\label{ch:v10-alpha-pi4-inverse-response}
Пусть два нормированных граничных сопряжения равны
\begin{equation}g_h=\frac{r_h}{S_{geo}},\qquad g_c=\frac{r_c}{S_{geo}},
\qquad r_hr_c=\pi^{-4}.\end{equation}
Наиболее общая минимальная квадратичная диссипативная поправка имеет вид
\begin{equation}\Delta S_{resp}=-\chi_{resp}g_hg_c
=-\frac{\chi_{resp}}{\pi^4S_{geo}^2},\qquad \chi_{resp}>0.\end{equation}
Положительность восприимчивости фиксирует знак вычитания, а совместное
масштабирование $g_h,g_c\mapsto t g_h,t g_c$ даёт нулевую первую и ненулевую
вторую производную при $t=0$. Поэтому знак и порядок $S_{geo}^{-2}$ следуют
условно из квадратичного обратного отклика.

Но значение $\chi_{resp}$ не фиксируется KMS-отношениями. При
$\chi_{resp}=1$ получается поздняя формула
$-1/(\pi^4S_{geo}^2)$, тогда как столь же положительный выбор
$\chi_{resp}=\pi^3/32$ точно воспроизводит раннюю формулу
$-1/(32\pi S_{geo}^2)$. Шесть кандидатов происхождения восприимчивости дают
$0/6$ полных проходов; наблюдаемый отклик имеет ранг/ядро $1/2$.

\begin{theorem}[Условная структура третьей поправки]
Положительный двухрезервуарный cross-response вместе с
$r_hr_c=\pi^{-4}$ даёт отрицательную поправку второго порядка
$-\chi_{resp}/(\pi^4S_{geo}^2)$.
\end{theorem}
\begin{nogocriterion}[Неидентифицируемый коэффициент]
Поскольку единичная и историческая $32\pi$-нормировки принадлежат одной
положительной семье, третий член не выведен, пока отдельный родитель не
зафиксирует $\chi_{resp}=1$ и морфизм отклика в $\alpha^{-1}$.
\end{nogocriterion}
\begin{protocol}\begin{enumerate}\item условная структура: \textbf{$4/4$};
\item кандидаты происхождения: \textbf{$0/6$};\item коэффициент $1$: \textbf{$0/1$};
\item исключение $32\pi$: \textbf{$0/1$};\item физическое происхождение третьей поправки: \textbf{$0/1$};
\item следующий гейт: \textbf{аудит родителя cross-восприимчивости}.\end{enumerate}\end{protocol}
